=====================================================================
 B. APPLICATION  -  F407G-Application
=====================================================================

B1. CubeMX (F407G-Application.ioc):
    - Chan PA0 -> GPIO_EXTI0
    - GPIO mode: External Interrupt Mode with Rising edge trigger detection
    - Tab NVIC: tick "EXTI line0 interrupt"
    - Generate Code
    (Neu board cua ban noi nut User o chan khac thi doi PA0 cho dung.)

B2. Them app_ota.c vao Core/Src, app_ota.h vao Core/Inc.

B3. main.c cua Application:

/* USER CODE BEGIN Includes */
#include "app_header.h"
#include "flash_layout.h"
#include "app_ota.h"
/* USER CODE END Includes */

/* USER CODE BEGIN PV */
volatile uint8_t g_btn_pressed = 0U;
static uint32_t  s_last_press_tick = 0U;
/* ... giu nguyen khai bao app_header const o day ... */
/* USER CODE END PV */

/* USER CODE BEGIN 4 */
void HAL_GPIO_EXTI_Callback(uint16_t GPIO_Pin)
{
    if (GPIO_Pin == GPIO_PIN_0)                       /* nut User B1 = PA0 */
    {
        uint32_t now = HAL_GetTick();
        if ((now - s_last_press_tick) >= 50U)         /* debounce 50 ms */
        {
            s_last_press_tick = now;
            g_btn_pressed = 1U;                       /* ISR chi dat co */
        }
    }
}
/* USER CODE END 4 */

/* trong while (1)  -  USER CODE BEGIN 3 */
    if (g_btn_pressed != 0U)
    {
        g_btn_pressed = 0U;
        if (app_ota_request() == 0)
        {
            NVIC_SystemReset();
        }
    }
    HAL_GPIO_TogglePin(APP_LED_PORT, APP_LED_PIN);
    HAL_Delay(100);
